建表与增删改查

          
最小闭环：建表 → 插数据 → 查 → 改 → 删。表建在 demo 上，不碰示例数据。

          
CREATE TABLEINSERTSELECTUPDATEDELETE

          
DROP TABLE IF EXISTS demo;
CREATE TABLE demo (id serial PRIMARY KEY, name text, n int);
INSERT INTO demo (name, n) VALUES ('a', 1), ('b', 2);
SELECT * FROM demo ORDER BY id;
UPDATE demo SET n = n + 1 WHERE name = 'a';
DELETE FROM demo WHERE name = 'b';

查询：筛选、排序、分页、去重

          
日常用得最多的一层，条件、排序、翻页、去重都在这里。

          
WHEREAND / OR / NOTINBETWEENLIKE / ILIKEIS NULLORDER BY ASC/DESC NULLS FIRST|LASTLIMIT / OFFSETDISTINCT

          
SELECT DISTINCT city FROM departments ORDER BY city;

SELECT name, salary FROM employees
WHERE salary BETWEEN 20000 AND 45000 AND name LIKE '%张%'
ORDER BY salary DESC
LIMIT 5;

        

      


      

        
3

        

          
分组统计

          
聚合函数配 GROUP BY；分组后再筛选用 HAVING（不是 WHERE）。

          
count / sum / avg / min / max（含 DISTINCT）· GROUP BYHAVING

          
SELECT dept_id, count(*) AS n, round(avg(salary), 2) AS avg_pay
FROM employees
GROUP BY dept_id
HAVING count(*) > 1
ORDER BY n DESC;

        

      


      

        
4

        

          
多表连接

          
LEFT JOIN 保留左表全部行，对不上的右表列为 NULL —— 统计「没人/没单」时靠它。

          
INNER / LEFT / RIGHT / CROSS JOINON表别名与列别名括号联接组 (a JOIN b ON …) x

          
SELECT d.name, count(e.id) AS n
FROM departments d
LEFT JOIN employees e ON e.dept_id = d.id
GROUP BY d.name
ORDER BY n DESC;

        

      


      

        
5

        

          
子查询与集合运算

          
标量子查询返回单值，可以引用外层列；WITH 把中间结果命名一次，多处复用。

          
标量子查询IN (子查询)EXISTSFROM 派生表 (SELECT …) xVALUES (…),(…)UNION [ALL] / INTERSECT / EXCEPTWITH

          
SELECT name FROM employees
WHERE dept_id IN (SELECT id FROM departments WHERE budget > 100000)
  AND salary > (SELECT avg(salary) FROM employees);

WITH big AS (SELECT id FROM departments WHERE budget > 100000)
SELECT count(*) FROM employees WHERE dept_id IN (SELECT id FROM big);

        

      


      

        
6

        

          
写入进阶：返回、冲突、批量

          
ON CONFLICT 是「有就改、没有就插」；RETURNING 直接拿到写后的行，省一次查询。写在 demo_up 上。

          
RETURNINGON CONFLICT DO NOTHING | DO UPDATE SET … EXCLUDEDINSERT … SELECTUPDATE … FROMDELETE … USING写语句的 WHERE/SET/VALUES 里也能用子查询

          
DROP TABLE IF EXISTS demo_up;
CREATE TABLE demo_up (id int PRIMARY KEY, name text, salary numeric(10,2));

INSERT INTO demo_up VALUES (1, '张伟', 42000)
ON CONFLICT (id) DO UPDATE SET salary = EXCLUDED.salary
RETURNING id, name, salary;

UPDATE demo_up d SET salary = s.salary
FROM (VALUES (1, 50000)) s(id, salary)
WHERE d.id = s.id
RETURNING d.id, d.salary;

表结构与索引

          
索引是内存 B+Tree，建完立刻生效；等值、范围、IN、ORDER BY 免排序都会用到。结构改动在 demo_ddl 上。

          
ALTER TABLE ADD / DROP / RENAME COLUMNRENAME TOALTER COLUMN TYPECREATE [UNIQUE] INDEX [IF NOT EXISTS]DROP INDEXPRIMARY KEY / UNIQUE / NOT NULL / DEFAULT / CHECK / FOREIGN KEYGENERATED AS IDENTITYCOMMENT ONTRUNCATE

          
DROP TABLE IF EXISTS demo_ddl;
CREATE TABLE demo_ddl (id int PRIMARY KEY, dept_id int, note text);
ALTER TABLE demo_ddl ADD COLUMN email text;
CREATE INDEX IF NOT EXISTS idx_demo_ddl_dept ON demo_ddl (dept_id);
COMMENT ON TABLE demo_ddl IS '结构演示';
TRUNCATE TABLE demo_ddl;

事务与并发

          
表级 S/X 锁、严格两阶段封锁（可串行化）；SAVEPOINT 用于部分回滚，ORM 的嵌套事务靠它。演示在 demo_tx 上。

          
BEGIN / START TRANSACTION（ISOLATION LEVEL …、READ ONLY）· COMMITROLLBACKSAVEPOINT / ROLLBACK TO SAVEPOINT / RELEASE SAVEPOINT

          
DROP TABLE IF EXISTS demo_tx;
CREATE TABLE demo_tx (id int PRIMARY KEY, v int);
INSERT INTO demo_tx VALUES (1, 100), (2, 200);

BEGIN;
UPDATE demo_tx SET v = v + 1 WHERE id = 1;
SAVEPOINT sp1;
DELETE FROM demo_tx WHERE id = 2;
ROLLBACK TO SAVEPOINT sp1;   -- 只撤销上面那条 DELETE
COMMIT;

SELECT * FROM demo_tx ORDER BY id;

常用函数与类型

          
够覆盖大部分报表写法；序列函数只在 SERIAL 建出的序列上有效。interval 支持时间加减（月/年按日历精确计算）。

          
字符串 lower / upper / length / trim / lpad / rpad / replace / substring / position / concat / string_agg日期 extract / date_part / date_trunc / now · INTERVAL '1 day' 与时间加减序列 nextval / currval / lastval其他 coalesce / nullif / greatest / least / random / regexp_matches / unnest类型 int / bigint / serial / numeric(p,s) / real / double precision / text / varchar(n) / boolean / date / timestamp / interval

          
SELECT extract(year FROM now()) AS y,
       date_trunc('month', now()) AS m,
       lpad('1', 6, '0') AS code,
       now() - INTERVAL '1 day' AS 昨天,
       salary
FROM employees
ORDER BY salary DESC LIMIT 5;

SELECT unnest(array[10, 20, 30]) AS n;   -- 数组展开成多行
